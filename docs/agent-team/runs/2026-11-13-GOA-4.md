     node --input-type=module -e "
     import { readJsonc } from './scripts/lib/jsonc.mjs';
     import { readFileSync, existsSync } from 'node:fs';
     const schemaOf = (file) => JSON.parse(readFileSync(file, 'utf8').match(/\{%-?\s*schema\s*-?%\}([\s\S]*?)\{%-?\s*endschema\s*-?%\}/)[1]);

     let bad = 0; const fail = (...a) => { bad++; console.log('FEL', ...a); };
     function checkSettings(where, file, settings = {}) {
       const defs = new Map((schemaOf(file).settings ?? []).filter((s) => s.id).map((s) => [s.id, s]));
       for (const [id, value] of Object.entries(settings)) {
         const def = defs.get(id);
         if (!def) { fail(where, 'okänd inställning', id); continue; }
         if (def.type === 'select' && !def.options.some((o) => o.value === value)) fail(where, id, 'ogiltigt värde', value);
         if (def.type === 'range' && (value < def.min || value > def.max)) fail(where, id, 'utanför intervallet', value);
         if (EMPTY.has(def.type) && value !== '' && !(Array.isArray(value) && value.length === 0)) fail(where, id, 'ska vara tom');
         if (def.type === 'collection' && value !== '' && value !== 'all') fail(where, id, 'hårdkodad kollektion', value);
       }
     }
     function checkBlocks(where, node) {
       const blocks = node.blocks ?? {};
       for (const id of node.block_order ?? []) if (!blocks[id]) fail(where, 'block_order pekar på okänt block', id);
       for (const [id, block] of Object.entries(blocks)) {
         const file = 'blocks/' + block.type + '.liquid';
         if (!existsSync(file)) { fail(where, id, 'blocktyp saknas', block.type); continue; }
         checkSettings(where + ' > ' + id, file, block.settings);
         checkBlocks(where + ' > ' + id, block);
       }
     }
     for (const id of tpl.order) if (!tpl.sections[id]) fail('order pekar på okänd sektion', id);
     for (const [id, section] of Object.entries(tpl.sections)) {
       const file = 'sections/' + section.type + '.liquid';
       if (!existsSync(file)) { fail(id, 'sektionstyp saknas', section.type); continue; }
       checkSettings(id, file, section.settings);
       checkBlocks(id, section);
     }
     console.log(tpl.order.map((id) => id + ':' + tpl.sections[id]?.type).join(' '));
     console.log(bad ? 'FAIL' : 'PASS'); process.exit(bad ? 1 : 0);
     " templates/index.json

## Risker

1. **`theme pull` skriver över `templates/index.json`.** Filen ägs av theme editor så snart någon redigerar startsidan i butiken. Hämtas temat innan den här ändringen är uppladdad försvinner de nya sektionerna ur repot. Ordningen måste vara: merge och push först, redigering i editorn sedan, `theme pull` sist. `HANDOFF.md` säger redan att push ska ske före pull.
2. **`theme push` skriver över editorns startsida.** Laddas branchen upp till ett tema där någon redan ställt in startsidan (till exempel `Parfymia GOA-3`, id `162800009301`) ersätts den inställningen. Människan avgör vilket tema som är mål.
3. **Inställningar som inte finns i schemat.** En okänd nyckel eller ett ogiltigt `select`-värde kan göra att Shopify avvisar mallen eller tyst släpper värdet. `verify.sh` fångar inte detta (bara JSON-syntax). Därför strukturkontrollen i T2.
4. **Platshållare tills människan valt bilder.** Bildblock utan bild visar Horizons klädplatshållare (`blocks/image.liquid` rad 56-60), hero likaså (`sections/hero.liquid` rad 329-331). Startsidan ser alltså ofärdig ut direkt efter push. Bildblock utan länk är inte klickbara (rad 63-70).
5. **Mobil-layouten är generisk flex.** Kategorier och logotyper ligger i en `group` med rad, och bryts på mobil via `--flex-wrap-mobile: wrap` (`snippets/layout-panel-style.liquid` rad 28, `assets/base.css` rad 2610-2612). Bredderna 45 % och 28 % på mobil och 12 % på dator är planerarens förslag och är inte renderade. Fem kategorier ger två, två, en på mobil. Blir det inte bra justeras värdena i theme editor. Går det inte att få bra där är `sections/collection-list.liquid` reservalternativet (Kräver godkännande 6).

## F1 och rättningen

- **Fel:** kategorigrupperna hade bredd "fyll" och egen mobilbredd 45 %. "Fyll" ger `flex: 1` (`assets/base.css`, `.group-block--width-fill`), som går före mobilbredden. Raden skulle brytas efter ordens bredd och sista rutan sträckas ut.
- **Rättning:** `width: custom`, `custom_width: 28`, `width_mobile: custom`, `custom_width_mobile: 28` på `category_1` till `category_5`.
- **Följd enligt kodläsning:** dator, fem lika rutor som krymper till en rad. Mobil vid 390 och 320 px, 3 + 2 med lika stora rutor, centrerade.
- **Att känna till:** i en rad styrs gruppens bredd av `flex-basis`, som tar värdet från datorfältet "egen bredd". Mobilfältet påverkar inte radbrytningen. Vill handlaren ha två per rad på mobil ändras datorfältet till 45; på dator krymper rutorna ändå till en rad.
- Rättningen gjordes av orkestratorn, inte av `theme-developer`. Den granskades oberoende.
